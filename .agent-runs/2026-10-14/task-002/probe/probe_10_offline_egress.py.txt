"""probe_10_offline_egress.py — 验证 langgraph 在「零外网」下是否安全

背景：langchain-core 依赖 langsmith（langgraph 依赖 langchain-core）。
langsmith 是追踪/遥测组件，理论上可能联网。项目约束「零外网、运行期外发请求 = 0」。
做法：在导入 langgraph 及跑一次图之前，给 socket.socket.connect / connect_ex 打补丁，
记录所有连接尝试（含地址判定），看是否有非 localhost 的外发。
"""
import socket
import sys

ATTEMPTS = []
_orig_connect = socket.socket.connect
_orig_connect_ex = socket.socket.connect_ex


def _is_local(addr):
    try:
        host = addr[0]
    except Exception:  # noqa: BLE001
        return False
    return host in ("127.0.0.1", "::1", "localhost", "0.0.0.0")


def _patched_connect(self, addr):
    ATTEMPTS.append(("connect", addr))
    print(f"  [SOCKET] connect {addr!r}  local={_is_local(addr) if isinstance(addr, tuple) else '?'}")
    return _orig_connect(self, addr)


def _patched_connect_ex(self, addr):
    ATTEMPTS.append(("connect_ex", addr))
    print(f"  [SOCKET] connect_ex {addr!r}")
    return _orig_connect_ex(self, addr)


socket.socket.connect = _patched_connect
socket.socket.connect_ex = _patched_connect_ex
socket.create_connection = lambda *a, **k: (_ for _ in ()).throw(
    AssertionError(f"create_connection 被调用：{a} {k}"))

print("=" * 70)
print("1) 导入链中是否引入 requests / langsmith / socket 相关模块")
print("=" * 70)
import langgraph  # noqa: E402
from langgraph.graph import StateGraph, START, END  # noqa: E402
from langgraph.checkpoint.memory import MemorySaver  # noqa: E402
from langgraph.types import interrupt, Command  # noqa: E402

for mod in ["langsmith", "requests", "urllib.request", "http.client", "socket",
            "langchain_core", "langgraph", "httpx"]:
    print(f"  {mod:20s} in sys.modules = {mod in sys.modules}")

print()
print("=" * 70)
print("2) 跑一次带 interrupt + resume 的图，看是否有 socket 连接尝试")
print("=" * 70)
from typing import Annotated  # noqa: E402
from typing_extensions import TypedDict  # noqa: E402
from langgraph.graph.message import add_messages  # noqa: E402


class S(TypedDict):
    log: Annotated[list, add_messages]


def n(state: S):
    v = interrupt("ask")
    return {"log": [str(v)]}


b = StateGraph(S)
b.add_node("n", n)
b.add_edge(START, "n")
b.add_edge("n", END)
g = b.compile(checkpointer=MemorySaver())
cfg = {"configurable": {"thread_id": "off1"}}
g.invoke({"log": []}, config=cfg)
g.invoke(Command(resume="approve"), config=cfg)

print()
print("=" * 70)
print("3) 结论材料")
print("=" * 70)
print("  socket 连接尝试次数 =", len(ATTEMPTS))
print("  连接尝试明细       =", ATTEMPTS)
